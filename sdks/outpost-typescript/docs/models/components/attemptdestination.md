# AttemptDestination

The destination an attempt was sent to, without credentials. Use the destination endpoints to retrieve credentials.

## Example Usage

```typescript
import { AttemptDestination } from "@hookdeck/outpost-sdk/models/components";

let value: AttemptDestination = {
  id: "des_12345",
  tenantId: "tenant_123",
  type: "webhook",
  topics: "*",
  filter: {
    "data": {
      "amount": {
        "$gte": 100,
      },
      "customer": {
        "tier": "premium",
      },
    },
  },
  config: {
    "url": "https://my-service.com/webhook/handler",
  },
  deliveryMetadata: {
    "app-id": "my-app",
    "region": "us-east-1",
  },
  metadata: {
    "internal-id": "123",
    "team": "platform",
  },
  disabledAt: null,
  createdAt: new Date("2024-01-01T00:00:00Z"),
  updatedAt: new Date("2024-01-01T00:00:00Z"),
  target: "my-service.com",
  targetUrl: "https://my-service.com/webhook/handler",
};
```
